Mã Google Apps Script Tự Động Cập Nhật Số Liệu CMS Theo Tháng (Trang 3)

Thực hiện 3 bước đơn giản trên file Google Sheet của Trang 3 để tự động hóa 100%:

  1. Mở file Google Sheet Trang 3 (Sheet ID: 1Z70EDNp9lg1DPrAAUMmpMGf6f_rYEVZTZ2VqpRfpWpw), vào menu Tiện ích mở rộng (Extensions) > Apps Script.
  2. Xóa toàn bộ mã cũ trong file Code.gs, dán đoạn mã chuẩn hóa bên dưới vào rồi bấm Lưu (Ctrl + S).
  3. Bấm nút Triển khai (Deploy) > Tùy chọn triển khai mới (New deployment) > Chọn loại Ứng dụng web (Web app) > Mục "Người thực thi" (Execute as) chọn "Tôi" (Me) > Mục "Ai có quyền truy cập" (Who has access) chọn "Bất kỳ ai" (Anyone) (Rất quan trọng!) > Bấm Triển khai > Sao chép URL Web App dán vào ô bên dưới.
/**
 * GOOGLE APPS SCRIPT WEB APP - TỰ ĐỘNG CẬP NHẬT CHỈ SỐ CMS THEO THÁNG (TRANG 3)
 * Viện Quản Lý Chất Lượng & Khảo Thí (IQA)
 * Sheet ID: 1Z70EDNp9lg1DPrAAUMmpMGf6f_rYEVZTZ2VqpRfpWpw
 * Tab GID Mục Tiêu: 0
 */

function doPost(e) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var contents = e.postData ? e.postData.contents : "{}";
    var data = JSON.parse(contents);

    // 1. Tìm đúng tab có GID = 0 (hoặc tab đầu tiên)
    var targetGid = (data && data.gid !== undefined) ? Number(data.gid) : 0;
    var sheet = null;
    var sheets = ss.getSheets();
    for (var i = 0; i < sheets.length; i++) {
      if (sheets[i].getSheetId() == targetGid) {
        sheet = sheets[i];
        break;
      }
    }
    if (!sheet) sheet = ss.getActiveSheet() || sheets[0];

    // 2. Lấy thông tin Năm và Tháng cần cập nhật
    var targetYear = Number(data.year || 2026);
    var targetMonth = Number(data.month || 10);

    // 3. Quét cột A để tìm chính xác dòng của Năm và Tháng đó
    var lastRow = sheet.getLastRow();
    var colA = sheet.getRange(1, 1, lastRow, 1).getValues();
    var currentYear = 2025;
    var foundRow = -1;

    for (var r = 0; r < colA.length; r++) {
      var cellVal = String(colA[r][0] || "").trim();
      var yearMatch = cellVal.match(/(?:Năm\s*)(\d{4})/i);
      if (yearMatch) {
        currentYear = Number(yearMatch[1]);
      }
      var monthMatch = cellVal.match(/(?:Tháng\s*)(\d{1,2})/i);
      if (monthMatch) {
        var mNum = Number(monthMatch[1]);
        if (currentYear === targetYear && mNum === targetMonth) {
          foundRow = r + 1; // 1-indexed trong Google Sheet
          break;
        }
      }
    }

    // Dự phòng vị trí hàng cố định theo cấu trúc sheet
    if (foundRow === -1) {
      if (targetYear === 2025) foundRow = 1 + targetMonth;
      else if (targetYear === 2026) foundRow = 14 + targetMonth;
      else if (targetYear === 2027) foundRow = 27 + targetMonth;
      else if (targetYear === 2028) foundRow = 40 + targetMonth;
    }

    if (foundRow < 2) {
      throw new Error("Không tìm thấy dòng cho Tháng " + targetMonth + " Năm " + targetYear);
    }

    // 4. Lấy 8 giá trị tương ứng 8 cột:
    // Cột B (2): Ảnh | Cột C (3): Bài đã đăng | Cột D (4): Bài chờ đăng | Cột E (5): Báo cáo
    // Cột F (6): Thông báo | Cột G (7): Danh mục | Cột H (8): Nhân sự | Cột I (9): Trang/Phân hệ
    var img = Number(data.images !== undefined ? data.images : 0);
    var pub = Number(data.publishedPosts !== undefined ? data.publishedPosts : 0);
    var pen = Number(data.pendingPosts !== undefined ? data.pendingPosts : 0);
    var rep = Number(data.reports !== undefined ? data.reports : 0);
    var ann = Number(data.announcements !== undefined ? data.announcements : 0);
    var cat = Number(data.categories !== undefined ? data.categories : 0);
    var stf = Number(data.staff !== undefined ? data.staff : 0);
    var pag = Number(data.pages !== undefined ? data.pages : 0);

    // Ghi đè 8 cột từ B đến I tại dòng foundRow
    sheet.getRange(foundRow, 2, 1, 8).setValues([[img, pub, pen, rep, ann, cat, stf, pag]]);

    return ContentService.createTextOutput(JSON.stringify({
      status: "success",
      message: "Đã cập nhật thành công Tháng " + targetMonth + " Năm " + targetYear + " tại dòng " + foundRow + " trên Google Sheet!",
      row: foundRow
    })).setMimeType(ContentService.MimeType.JSON);

  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({
      status: "error",
      message: err.toString()
    })).setMimeType(ContentService.MimeType.JSON);
  }
}

function doGet(e) {
  return ContentService.createTextOutput("Webhook Google Sheet Trang 3 (CMS & Hoạt động) đã sẵn sàng!");
}
